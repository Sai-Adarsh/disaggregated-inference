cmake_minimum_required(VERSION 3.21)
project(disaggregated_inference LANGUAGES CXX)

option(ENABLE_RDMA "Enable librdmacm rsockets" OFF)
option(ENABLE_CUDA "Enable CUDA attention fixture" OFF)
find_package(Threads REQUIRED)
add_executable(split_inference main.cpp model.cpp transport.cpp)
target_compile_features(split_inference PRIVATE cxx_std_20)
target_link_libraries(split_inference PRIVATE Threads::Threads)

if(ENABLE_RDMA)
    find_path(RSOCKET_INCLUDE_DIR rdma/rsocket.h REQUIRED)
    find_library(RDMACM_LIBRARY rdmacm REQUIRED)
    target_include_directories(split_inference PRIVATE "${RSOCKET_INCLUDE_DIR}")
    target_link_libraries(split_inference PRIVATE "${RDMACM_LIBRARY}")
    target_compile_definitions(split_inference PRIVATE INFERENCE_RDMA)
endif()

if(ENABLE_CUDA)
    enable_language(CUDA)
    find_package(CUDAToolkit REQUIRED)
    target_sources(split_inference PRIVATE model_cuda.cu)
    target_compile_features(split_inference PRIVATE cuda_std_17)
    target_link_libraries(split_inference PRIVATE CUDA::cudart)
    target_compile_definitions(split_inference PRIVATE INFERENCE_CUDA)
endif()
